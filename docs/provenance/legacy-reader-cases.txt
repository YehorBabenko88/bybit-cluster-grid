class Connection:
 def __init__(self,records=(),active=0):self.info=SimpleNamespace(transaction_status=active);self.records=records;self.calls=[];self.in_tx=False
 @contextmanager
 def transaction(self):
  self.in_tx=True
  try:yield
  finally:self.in_tx=False
 @contextmanager
 def cursor(self):yield self
 def execute(self,sql,args=None):assert self.in_tx;self.calls.append((sql,args))
 def fetchall(self):return self.records

def test_readonly_transaction_must_retire_before_return_without_repair():
    from contextlib import contextmanager
    class Connection:
        info=SimpleNamespace(transaction_status=0)
        @contextmanager
        def transaction(self):
            yield self
            self.info.transaction_status=2
        @contextmanager
        def cursor(self):yield self
        def execute(self,*args):pass
        def fetchall(self):return []
    q=dict(symbol='BTCUSDT',after_id=0,start_ms=0,end_ms=1,as_of_ms=0,max_age_ms=1)
    conn=Connection()
    with pytest.raises(module.ExportBlocked):module.read_paper_inputs(conn,q)
    assert conn.info.transaction_status==2
